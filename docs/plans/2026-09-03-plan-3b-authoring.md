## Deviations from the spec (decided while grounding the plan in the code — record, don't re-litigate)

1. **No new `media`/`callout` block types.** The closed prose schema (`packages/schemas/prose-doc.schema.json`) and `ProseNode.svelte` ALREADY support `image` (attrs `mediaAssetId`+`alt`, rendered via the `prose-images` context with an alt-text fallback when the id resolves to nothing), `callout`, `math`, and `table` INSIDE `rich_text` bodies. So images and callouts live inside rich text; `BLOCK_TYPES` stays `("rich_text", "knowledge_check")` and needs no migration. The converter's img placeholder is a prose `image` node with a deterministic UUID that resolves to nothing (renders as alt text) until the author attaches a real asset.
2. **Question search has no `subject` filter** — the `question` table has no subject column. `GET /authoring/questions` filters by free-text `q` (ILIKE over `stem::text`) only.
3. **Math nodes are preserved, not authored.** The editor round-trips existing `math` nodes (rendered as source in a `<code>`-style chip) but has no insert-math UI in 3b.
4. **Editing lesson pages recreates their bank Question rows** (the importer's existing replace semantics, reused). The stable identifier is the knowledge-check block `key`; grading and analytics key on it / on pinned snapshots, so question-row churn is harmless (same situation as re-import, noted in plan 3a).

```python
def _correct_answers(soup: BeautifulSoup, notes: list[str]) -> dict[str, str]:
    """Gather `{key: correct_value}` from every encoding the legacy pages use: the
    first `lessonCorrectAnswers`/`correctAnswers` object literal (parsed with pyjson5
    for JS-literal syntax), then any `checkPageAnswer('<key>', '<value>')` onclick
    args (which never override an object-literal entry). Keys may or may not carry
    the radio name's `_ans` suffix; `_build_knowledge_check` tries both."""
    answers: dict[str, str] = {}
    for script in soup.find_all("script"):
        match = _CORRECT_ANSWERS_RE.search(script.get_text())
        if match:
            try:
                answers.update(pyjson5.decode(match.group(1)))
            except Exception:
                notes.append("unparseable lessonCorrectAnswers object")
            break
    for tag in soup.find_all(onclick=True):
        m = _CHECK_PAGE_ANSWER_RE.search(str(tag["onclick"]))
        if m and m.group(1) not in answers:
            answers[m.group(1)] = m.group(2)
    return answers
```

```python
from datetime import timedelta
from typing import Protocol
from urllib.parse import urlparse

from minio import Minio
from minio.error import S3Error

from app.config import get_settings


class MediaStorage(Protocol):
    def presigned_put(self, key: str) -> str: ...
    def presigned_get(self, key: str) -> str: ...
    def stat(self, key: str) -> int | None: ...


class MinioStorage:
    """S3-compatible storage via the minio SDK (endpoint/creds from Settings)."""

    def __init__(self) -> None:
        settings = get_settings()
        parsed = urlparse(settings.s3_endpoint)
        self._client = Minio(
            parsed.netloc,
            access_key=settings.s3_access_key,
            secret_key=settings.s3_secret_key,
            secure=parsed.scheme == "https",
        )
        self._bucket = settings.s3_bucket

    def presigned_put(self, key: str) -> str:
        return self._client.presigned_put_object(self._bucket, key, expires=timedelta(minutes=10))

    def presigned_get(self, key: str) -> str:
        return self._client.presigned_get_object(self._bucket, key, expires=timedelta(minutes=5))

    def stat(self, key: str) -> int | None:
        try:
            return self._client.stat_object(self._bucket, key).size
        except S3Error:
            return None


def get_media_storage() -> MediaStorage:
    return MinioStorage()
```

```python
    asset_id: uuid.UUID,
    request: Request,
    user: User = Depends(require_author),
    db: AsyncSession = Depends(get_db),
    storage: MediaStorage = Depends(get_media_storage),
) -> MediaAsset:
    """Verify the object landed in storage and mark the asset servable (audited)."""
    asset = await db.get(MediaAsset, asset_id)
    if asset is None:

    size = await anyio.to_thread.run_sync(storage.stat, asset.storage_key)
    if size is None:

    asset.bytes = size
    asset.confirmed = True
    await record_audit(
        db, actor=user, action="confirm_media", target_type="media_asset",
        target_id=asset.id, request=request,
    )
    await db.flush()
    return asset
```

```typescript
export function toEditor(node: ProseNode): JSONContent {
	if (node.type === 'tableCell' && node.attrs?.header) {
		return { type: 'tableHeader', content: node.content?.map(toEditor) };
	}
	const out: JSONContent = { ...node };
	if (node.content) out.content = node.content.map(toEditor);
	return out;
}

export function fromEditor(node: JSONContent): ProseNode {
	if (node.type === 'tableHeader') {
		return { type: 'tableCell', attrs: { header: true }, content: (node.content ?? []).map(fromEditor) };
	}
	if (node.type === 'tableCell') {
		// TipTap decorates cells with colspan/rowspan/colwidth; our schema has no cell attrs.
		return { type: 'tableCell', content: (node.content ?? []).map(fromEditor) };
	}
	const out: ProseNode = { type: node.type };
	// Allow-list rebuild: TipTap decorates nodes with editor-only attrs (empty attrs
	// objects, colwidth, etc.) the closed schema rejects — copy over ONLY what the schema
	// knows: attrs (when non-empty after dropping null/undefined values), marks (when
	// non-empty), text, content.
	if (node.attrs) {
		const attrs = Object.fromEntries(
			Object.entries(node.attrs).filter(([, v]) => v !== null && v !== undefined)
		);
		if (Object.keys(attrs).length > 0) out.attrs = attrs;
	}
	if (node.marks && node.marks.length > 0) out.marks = node.marks as ProseMark[];
	if (node.text !== undefined) out.text = node.text;
	if (node.content) out.content = node.content.map(fromEditor);
	return out;
}
```

If the round-trip test or the ajv check still finds TipTap-emitted shapes this allow-list
lets through (e.g. a default attr the schema lacks on headings/lists), extend the filter for
that node type specifically — iterate until BOTH the deep-equal round-trip and ajv validation
of `fromEditor` output pass.
